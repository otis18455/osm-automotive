Reserve: nummerplaat opzoeken

Dit bestand is een bewaarkopie van het nummerplaat-onderdeel dat uit het formulier is gehaald. Het doet niets op de website.

Weer aanzetten: plak het HTML-blok hieronder terug in verkopen.html (bovenaan in de fieldset "Uw auto", vóór <div class="row">) en het JavaScript-blok in verkopen.js (vóór "// validatie"). Zet het bestand api/kenteken.php met een api/config.php (Inmotiv-sleutel) online op hosting met PHP. De opmaak staat nog in styles.css.

1. HTML (verkopen.html)

<div class="fld">
  <label for="kenteken">Nummerplaat</label>
  <div class="plate-row">
    <div class="plate"><span class="eu">B</span><input id="kenteken" name="kenteken" maxlength="9" placeholder="1-ABC-123" autocapitalize="characters" autocomplete="off" spellcheck="false" inputmode="text"></div>
    <button type="button" class="btn btn-ghost" id="plateBtn">Gegevens ophalen</button>
  </div>
  <div class="plate-msg" id="plateMsg" role="status"></div>
</div>

2. JavaScript (verkopen.js)

// ---------- nummerplaat opzoeken (België, via api/kenteken.php) ----------
const plateIn = document.querySelector('#kenteken'), plateMsg = document.querySelector('#plateMsg'), plateBtn = document.querySelector('#plateBtn');
const say = (t, cls = '') => { plateMsg.textContent = t; plateMsg.className = 'plate-msg ' + cls; };
const cleanPlate = v => v.toUpperCase().replace(/[^A-Z0-9]/g, '');
const validPlate = p => /^([1-9][A-Z]{3}\d{3}|[A-Z]{3}\d{3}|\d{3}[A-Z]{3})$/.test(p);

// zoekt de eerste waarde onder één van de sleutelnamen, overal in het antwoord
function pick(obj, keys) {
  const want = keys.map(k => k.toLowerCase());
  let found;
  (function walk(o) {
    if (found !== undefined || o === null || typeof o !== 'object') return;
    for (const [k, v] of Object.entries(o)) {
      if (want.includes(k.toLowerCase()) && v !== null && typeof v !== 'object' && String(v).trim() !== '') { found = v; return; }
    }
    for (const v of Object.values(o)) walk(v);
  })(obj);
  return found;
}
function setSelect(sel, value) {
  if (!value) return false;
  const v = String(value).toLowerCase();
  const opt = [...sel.options].find(o => o.value && (o.value.toLowerCase() === v || v.includes(o.value.toLowerCase())));
  if (opt) { sel.value = opt.value; return true; }
  return false;
}
function fillFromLookup(d) {
  const merk = pick(d, ['make', 'brand', 'merk', 'makename']);
  const model = pick(d, ['model', 'modelname', 'modelo']);
  const datum = pick(d, ['firstregistrationdate', 'first_registration_date', 'datefirstregistration', 'firstregistration', 'eersteinschrijving', 'registrationdate']);
  const jaar = pick(d, ['year', 'buildyear', 'bouwjaar', 'constructionyear', 'modelyear']) || (datum && String(datum).match(/(19|20)\d{2}/)?.[0]);
  const fuel = String(pick(d, ['fuel', 'fueltype', 'brandstof', 'fuel_type']) || '').toLowerCase();
  const trans = String(pick(d, ['transmission', 'transmissiontype', 'gearbox', 'transmissie']) || '').toLowerCase();
  const kleur = pick(d, ['color', 'colour', 'kleur']);

  let filled = 0;
  if (merk) { form.merk.value = merk; filled++; }
  if (model) { form.model.value = model; filled++; }
  if (jaar && setSelect(form.bouwjaar, jaar)) filled++;
  const fuelMap = [['plug', 'Plug-in hybride'], ['hybri', 'Hybride'], ['diesel', 'Diesel'], ['electr', 'Elektrisch'], ['elektr', 'Elektrisch'], ['lpg', 'LPG / CNG'], ['cng', 'LPG / CNG'], ['gasol', 'Benzine'], ['petrol', 'Benzine'], ['benz', 'Benzine']];
  const f = fuelMap.find(([k]) => fuel.includes(k));
  if (f && setSelect(form.brandstof, f[1])) filled++;
  if (trans) { form.transmissie.value = /auto|cvt|dsg|dct/.test(trans) ? 'Automaat' : 'Handgeschakeld'; filled++; }
  if (kleur) { form.kleur.value = kleur; filled++; }
  return filled;
}

plateIn.addEventListener('input', () => {
  plateIn.value = plateIn.value.toUpperCase();
  say('');
});
// Beperkt het aantal opzoekingen (elke opzoeking kost geld)
const MAX_LOOKUPS = 3;
const lookedUp = new Set();
let blocked = false;
let lookups = Number(sessionStorage.getItem('lookups') || 0);
plateIn.addEventListener('input', () => { if (!lookedUp.has(cleanPlate(plateIn.value))) plateBtn.disabled = blocked || lookups >= MAX_LOOKUPS; });
async function lookup() {
  const p = cleanPlate(plateIn.value);
  if (!validPlate(p)) return say('Voer een geldige Belgische nummerplaat in, bijvoorbeeld 1-ABC-123.', 'bad');
  if (lookedUp.has(p)) return say('Deze nummerplaat hebt u al opgezocht.', 'bad');
  if (lookups >= MAX_LOOKUPS) { plateBtn.disabled = true; return say('U hebt het maximum aantal opzoekingen bereikt. Vul de gegevens zelf in.', 'bad'); }
  plateBtn.disabled = true; say('Gegevens opzoeken…');
  try {
    const res = await fetch('api/kenteken.php?plate=' + p);
    const json = await res.json().catch(() => ({}));
    if (!res.ok) {
      const m = { not_found: 'Geen voertuig gevonden voor deze nummerplaat. Controleer de plaat of vul de gegevens zelf in.', rate_limited: 'U hebt het maximum aantal opzoekingen bereikt. Vul de gegevens zelf in.', daily_limit: 'Automatisch opzoeken is vandaag niet meer beschikbaar. Vul de gegevens zelf in.', invalid_plate: 'Ongeldige nummerplaat.' };
      blocked = ['rate_limited', 'daily_limit'].includes(json.error);
      return say(m[json.error] || 'Automatisch opzoeken is nu niet beschikbaar. Vul de gegevens hieronder zelf in.', 'bad');
    }
    lookedUp.add(p);
    if (!json.cached) { lookups++; sessionStorage.setItem('lookups', lookups); }
    const n = fillFromLookup(json.data);
    say(n ? 'Gegevens ingevuld. Controleer ze en vul aan waar nodig.' : 'Voertuig gevonden, maar we konden de velden niet invullen. Vul ze zelf in.', n ? 'ok' : 'bad');
  } catch {
    say('Automatisch opzoeken is nu niet beschikbaar. Vul de gegevens hieronder zelf in.', 'bad');
  } finally { plateBtn.disabled = blocked || lookedUp.has(p) || lookups >= MAX_LOOKUPS; }
}
plateBtn.addEventListener('click', lookup);
plateIn.addEventListener('keydown', e => { if (e.key === 'Enter') { e.preventDefault(); lookup(); } });
